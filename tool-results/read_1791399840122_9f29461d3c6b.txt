    55→    55→  }, []);
    56→    56→
    57→    57→  const stop = useCallback(() => {
    58→    58→    if (timerRef.current !== null) {
    59→    59→      window.clearInterval(timerRef.current);
    60→    60→      timerRef.current = null;
    61→    61→    }
    62→    62→  }, []);
    63→    63→
    64→    64→  const start = useCallback(() => {
    65→    65→    stop();
    66→    66→    const pattern = () => {
    67→    67→      beep(0, 754);
    68→    68→      beep(0.28, 754);
    69→    69→    };
    70→    70→    pattern();
    71→    71→    timerRef.current = window.setInterval(pattern, 2200);
    72→    72→  }, [beep, stop]);
    73→    73→
    74→    74→  useEffect(() => stop, [stop]);
    75→    75→
    76→    76→  return { start, stop };
    77→    77→}

    81→    81→  const s = sec % 60;
    82→    82→  return `${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    83→    83→}

    88→    88→  mirrored,

   107→   107→    if (ref.current && stream && ref.current.srcObject !== stream) {

   109→   109→    }
   110→   110→  }, [stream]);
   111→   111→
   112→   112→  const hasVideo = stream ? stream.getVideoTracks().some((t) => t.enabled && t.readyState === "live") : false;
   113→   113→

   148→   148→    </motion.div>
   149→   149→  );
   150→   150→}

   152→   152→export default function CallLayer() {
   153→   153→  const {
   154→   154→    me,
   155→   155→    call,
   156→   156→    incomingCall,
   157→   157→    conversations,
   158→   158→    callPeerStates,
   159→   159→    acceptIncomingCall,
   160→   160→    declineIncomingCall,
   161→   161→    cancelCall,
   162→   162→    leaveCall,
   163→   163→    endCallIfActive,
   164→   164→    setCallStatus,
   165→   165→    setPeerState,
   166→   166→  } = useAxStore();
   167→   167→  const { toast } = useToast();
   168→   168→  const ringer = useRinger();
   169→   169→
   170→   170→  const [localStream, setLocalStream] = useState<MediaStream | null>(null);
   171→   171→  const [remoteStreams, setRemoteStreams] = useState<Record<string, MediaStream>>({});
   172→   172→  const [remoteIds, setRemoteIds] = useState<string[]>([]);
   173→   173→  const [micOn, setMicOn] = useState(true);
   174→   174→  const [camOn, setCamOn] = useState(true);
   175→   175→  const [now, setNow] = useState(Date.now());
   176→   176→
   177→   177→  const pcsRef = useRef<Map<string, RTCPeerConnection>>(new Map());
   178→   178→  const pendingIceRef = useRef<Map<string, RTCIceCandidateInit[]>>(new Map());
   179→   179→  const localStreamRef = useRef<MediaStream | null>(null);
   180→   180→
   181→   181→  const conv = useMemo(
   182→   182→    () => conversations.find((c) => c.id === (call?.conversationId ?? incomingCall?.conversationId)),
   183→   183→    [conversations, call, incomingCall]
   184→   184→  );
   185→   185→
   186→   186→  const memberInfo = (id: string): AxUser | undefined =>
   187→   187→    conv?.members.find((m) => m.id === id);
   188→   188→
   189→   189→  // Reloj de duración
   190→   190→  useEffect(() => {
   191→   191→    if (!call) return;
   192→   192→    const t = window.setInterval(() => setNow(Date.now()), 1000);
   193→   193→    return () => window.clearInterval(t);
   194→   194→  }, [call]);
   195→   195→
   196→   196→  const cleanupMedia = useCallback(() => {
   197→   197→    for (const pc of pcsRef.current.values()) {
   198→   198→      try {
   199→   199→        pc.close();
   200→   200→      } catch {
   201→   201→        /* noop */
   202→   202→      }
   203→   203→    }
   204→   204→    pcsRef.current.clear();
   205→   205→    pendingIceRef.current.clear();
   206→   206→    localStreamRef.current?.getTracks().forEach((t) => t.stop());
   207→   207→    localStreamRef.current = null;
   208→   208→    setLocalStream(null);
   209→   209→    setRemoteStreams({});
   210→   210→    setRemoteIds([]);
   211→   211→    setMicOn(true);
   212→   212→    setCamOn(true);
   213→   213→  }, []);
   214→   214→
   215→   215→  // Adquirir medios al entrar en llamada
   216→   216→  useEffect(() => {
   217→   217→    if (!call) {
   218→   218→      cleanupMedia();
   219→   219→      return;
   220→   220→    }
   221→   221→    let cancelled = false;
   222→   222→    (async () => {
   223→   223→      try {
   224→   224→        const stream = await navigator.mediaDevices.getUserMedia({
   225→   225→          audio: true,
   226→   226→          video:
   227→   227→            call.type === "video"
   228→   228→              ? { width: { ideal: 1280 }, height: { ideal: 720 } }
   229→   229→              : false,
   230→   230→        });
   231→   231→        if (cancelled) {
   232→   232→          stream.getTracks().forEach((t) => t.stop());
   233→   233→          return;
   234→   234→        }
   235→   235→        localStreamRef.current = stream;
   236→   236→        setLocalStream(stream);
   237→   237→      } catch {
   238→   238→        // Sin micrófono/cámara: modo solo escucha con transceivers recvonly
   239→   239→        if (!cancelled) {
   240→   240→          localStreamRef.current = null;
   241→   241→          setLocalStream(null);
   242→   242→          toast({
   243→   243→            description:
   244→   244→              "Sin acceso a micrófono/cámara. Entrarás en modo solo escucha.",
   245→   245→          });
   246→   246→        }
   247→   247→      }
   248→   248→    })();
   249→   249→    return () => {
   250→   250→      cancelled = true;
   251→   251→    };
   252→   252→  }, [call?.conversationId]);
   253→   253→
   254→   254→  const createPeer = useCallback(
   255→   255→    async (remoteId: string, initiator: boolean, conversationId: string, video: boolean) => {
   256→   256→      const socket = getSocket();
   257→   257→      if (!socket || pcsRef.current.has(remoteId)) return;
   258→   258→
   259→   259→      const pc = new RTCPeerConnection(ICE_CONFIG);
   260→   260→      pcsRef.current.set(remoteId, pc);
   261→   261→      setRemoteIds((prev) => (prev.includes(remoteId) ? prev : [...prev, remoteId]));
   262→   262→
   263→   263→      pc.onicecandidate = (e) => {
   264→   264→        if (e.candidate) {
   265→   265→          socket.emit("call:ice", { conversationId, targetUserId: remoteId, candidate: e.candidate });
   266→   266→        }
   267→   267→      };
   268→   268→
   269→   269→      pc.ontrack = (e) => {
   270→   270→        if (e.streams[0]) {
   271→   271→          setRemoteStreams((prev) => ({ ...prev, [remoteId]: e.streams[0] }));
   272→   272→        }
   273→   273→      };
   274→   274→
   275→   275→      pc.onconnectionstatechange = () => {
   276→   276→        if (pc.connectionState === "connected") {
   277→   277→          setCallStatus("active");
   278→   278→        }
   279→   279→      };
   280→   280→
   281→   281→      const stream = localStreamRef.current;
   282→   282→      if (stream) {
   283→   283→        for (const track of stream.getTracks()) {
   284→   284→          pc.addTrack(track, stream);
   285→   285→        }
   286→   286→      } else {
   287→   287→        pc.addTransceiver("audio", { direction: "recvonly" });
   288→   288→        if (video) pc.addTransceiver("video", { direction: "recvonly" });
   289→   289→      }
   290→   290→
   291→   291→      if (initiator) {
   292→   292→        const offer = await pc.createOffer();
   293→   293→        await pc.setLocalDescription(offer);
   294→   294→        socket.emit("call:offer", {
   295→   295→          conversationId,
   296→   296→          targetUserId: remoteId,
   297→   297→          sdp: pc.localDescription,
   298→   298→        });
   299→   299→      }
   300→   300→    },
   301→   301→    [setCallStatus]
   302→   302→  );
   303→   303→
   304→   304→  const flushIce = useCallback(async (userId: string) => {
   305→   305→    const pc = pcsRef.current.get(userId);
   306→   306→    if (!pc || !pc.remoteDescription) return;
   307→   307→    const queue = pendingIceRef.current.get(userId) ?? [];
   308→   308→    pendingIceRef.current.set(userId, []);
   309→   309→    for (const candidate of queue) {
   310→   310→      try {
   311→   311→        await pc.addIceCandidate(candidate);
   312→   312→      } catch {
   313→   313→        /* candidato obsoleto */
   314→   314→      }
   315→   315→    }
   316→   316→  }, []);
   317→   317→
   318→   318→  // Listeners de señalización
   319→   319→  useEffect(() => {
   320→   320→    const socket = getSocket();
   321→   321→    if (!socket || !me) return;
   322→   322→
   323→   323→    const onIncoming = (data: { conversationId: string; type: "audio" | "video"; from: AxUser }) => {
   324→   324→      const state = useAxStore.getState();
   325→   325→      if (state.call) {
   326→   326→        // Ocupado: rechazar automáticamente
   327→   327→        socket.emit("call:decline", { conversationId: data.conversationId });
   328→   328→        return;
   329→   329→      }
   330→   330→      if (state.incomingCall?.conversationId === data.conversationId) return;
   331→   331→      useAxStore.setState({ incomingCall: data });
   332→   332→      ringer.start();
   333→   333→    };
   334→   334→
   335→   335→    const onPeerJoined = (data: { conversationId: string; userId: string }) => {
   336→   336→      const state = useAxStore.getState();
   337→   337→      if (state.call?.conversationId !== data.conversationId) return;
   338→   338→      void createPeer(data.userId, true, data.conversationId, state.call.type === "video");
   339→   339→    };
   340→   340→
   341→   341→    const onAccepted = (data: { conversationId: string; participantIds: string[] }) => {
   342→   342→      const state = useAxStore.getState();
   343→   343→      if (state.call?.conversationId !== data.conversationId) return;
   344→   344→      // Los participantes existentes nos ofrecerán conexión; solo registramos
   345→   345→      setRemoteIds((prev) => {
   346→   346→        const next = [...prev];
   347→   347→        for (const id of data.participantIds) {
   348→   348→          if (!next.includes(id)) next.push(id);
   349→   349→        }
   350→   350→        return next;
   351→   351→      });
   352→   352→    };
   353→   353→
   354→   354→    const onOffer = async (data: {
   355→   355→      conversationId: string;
   356→   356→      fromUserId: string;
   357→   357→      sdp: RTCSessionDescriptionInit;
   358→   358→    }) => {
   359→   359→      const state = useAxStore.getState();
   360→   360→      if (state.call?.conversationId !== data.conversationId) return;
   361→   361→      let pc = pcsRef.current.get(data.fromUserId);
   362→   362→      if (!pc) {
   363→   363→        await createPeer(
   364→   364→          data.fromUserId,
   365→   365→          false,
   366→   366→          data.conversationId,
   367→   367→          state.call.type === "video"
   368→   368→        );
   369→   369→        pc = pcsRef.current.get(data.fromUserId);
   370→   370→      }
   371→   371→      if (!pc) return;
   372→   372→      try {
   373→   373→        await pc.setRemoteDescription(data.sdp);
   374→   374→        await flushIce(data.fromUserId);
   375→   375→        const answer = await pc.createAnswer();
   376→   376→        await pc.setLocalDescription(answer);
   377→   377→        socket.emit("call:answer", {
   378→   378→          conversationId: data.conversationId,
   379→   379→          targetUserId: data.fromUserId,
   380→   380→          sdp: pc.localDescription,
   381→   381→        });
   382→   382→      } catch (e) {
   383→   383→        console.error("offer handling error", e);
   384→   384→      }
   385→   385→    };
   386→   386→
   387→   387→    const onAnswer = async (data: {
   388→   388→      conversationId: string;
   389→   389→      fromUserId: string;
   390→   390→      sdp: RTCSessionDescriptionInit;
   391→   391→    }) => {
   392→   392→      const pc = pcsRef.current.get(data.fromUserId);
   393→   393→      if (!pc) return;
   394→   394→      try {
   395→   395→        await pc.setRemoteDescription(data.sdp);
   396→   396→        await flushIce(data.fromUserId);
   397→   397→      } catch (e) {
   398→   398→        console.error("answer handling error", e);
   399→   399→      }
   400→   400→    };
   401→   401→
   402→   402→    const onIce = (data: { conversationId: string; fromUserId: string; candidate: RTCIceCandidateInit }) => {
   403→   403→      const state = useAxStore.getState();
   404→   404→      if (state.call?.conversationId !== data.conversationId) return;
   405→   405→      const pc = pcsRef.current.get(data.fromUserId);
   406→   406→      if (pc && pc.remoteDescription) {
   407→   407→        void pc.addIceCandidate(data.candidate).catch(() => undefined);
   408→   408→      } else {
   409→   409→        const q = pendingIceRef.current.get(data.fromUserId) ?? [];
   410→   410→        q.push(data.candidate);
   411→   411→        pendingIceRef.current.set(data.fromUserId, q);
   412→   412→      }
   413→   413→    };
   414→   414→
   415→   415→    const onPeerLeft = (data: { conversationId: string; userId: string }) => {
   416→   416→      const state = useAxStore.getState();
   417→   417→      if (state.call?.conversationId !== data.conversationId) return;
   418→   418→      const pc = pcsRef.current.get(data.userId);
   419→   419→      if (pc) {
   420→   420→        try {
   421→   421→          pc.close();
   422→   422→        } catch {
   423→   423→          /* noop */
   424→   424→        }
   425→   425→        pcsRef.current.delete(data.userId);
   426→   426→      }
   427→   427→      pendingIceRef.current.delete(data.userId);
   428→   428→      setRemoteStreams((prev) => {
   429→   429→        const next = { ...prev };
   430→   430→        delete next[data.userId];
   431→   431→        return next;
   432→   432→      });
   433→   433→      setRemoteIds((prev) => prev.filter((id) => id !== data.userId));
   434→   434→      const conv = state.conversations.find((c) => c.id === data.conversationId);
   435→   435→      const info = conv?.members.find((m) => m.id === data.userId);
   436→   436→      if (info) {
   437→   437→        toast({ description: `${info.displayName} salió de la llamada.` });
   438→   438→      }
   439→   439→      // En llamadas 1:1, si el otro cuelga, cerrar también la nuestra
   440→   440→      if (conv?.type === "dm") {
   441→   441→        useAxStore.setState({ call: null, callPeerStates: {} });
   442→   442→        // El servidor terminará la llamada y generará el mensaje del sistema
   443→   443→        const socket = getSocket();
   444→   444→        socket?.emit("call:leave", { conversationId: data.conversationId });
   445→   445→      }
   446→   446→    };
   447→   447→
   448→   448→    const onEnded = (data: { conversationId: string }) => {
   449→   449→      endCallIfActive(data.conversationId);
   450→   450→    };
   451→   451→
   452→   452→    const onDeclined = (data: { conversationId: string; userId: string; displayName: string }) => {
   453→   453→      const state = useAxStore.getState();
   454→   454→      if (state.call?.conversationId !== data.conversationId) return;
   455→   455→      toast({ description: `${data.displayName} rechazó la llamada.` });
   456→   456→    };
   457→   457→
   458→   458→    const onPeerState = (data: { conversationId: string; userId: string; audio?: boolean; video?: boolean }) => {
   459→   459→      const state = useAxStore.getState();
   460→   460→      if (state.call?.conversationId !== data.conversationId) return;
   461→   461→      setPeerState(data.userId, { audio: data.audio, video: data.video });
   462→   462→    };
   463→   463→
   464→   464→    const onBusy = (data: { conversationId: string }) => {
   465→   465→      const state = useAxStore.getState();
   466→   466→      if (state.call?.conversationId === data.conversationId) {
   467→   467→        toast({ description: "Ya hay una llamada en curso en este chat." });
   468→   468→        useAxStore.setState({ call: null });
   469→   469→      }
   470→   470→    };
   471→   471→
   472→   472→    socket.off("call:incoming", onIncoming);
   473→   473→    socket.on("call:incoming", onIncoming);
   474→   474→    socket.off("call:peer-joined", onPeerJoined);
   475→   475→    socket.on("call:peer-joined", onPeerJoined);
   476→   476→    socket.off("call:accepted", onAccepted);
   477→   477→    socket.on("call:accepted", onAccepted);
   478→   478→    socket.off("call:offer", onOffer);
   479→   479→    socket.on("call:offer", onOffer);
   480→   480→    socket.off("call:answer", onAnswer);
   481→   481→    socket.on("call:answer", onAnswer);
   482→   482→    socket.off("call:ice", onIce);
   483→   483→    socket.on("call:ice", onIce);
   484→   484→    socket.off("call:peer-left", onPeerLeft);
   485→   485→    socket.on("call:peer-left", onPeerLeft);
   486→   486→    socket.off("call:ended", onEnded);
   487→   487→    socket.on("call:ended", onEnded);
   488→   488→    socket.off("call:declined", onDeclined);
   489→   489→    socket.on("call:declined", onDeclined);
   490→   490→    socket.off("call:peer-state", onPeerState);
   491→   491→    socket.on("call:peer-state", onPeerState);
   492→   492→    socket.off("call:busy", onBusy);
   493→   493→    socket.on("call:busy", onBusy);
   494→   494→
   495→   495→    return () => {
   496→   496→      socket.off("call:incoming", onIncoming);
   497→   497→      socket.off("call:peer-joined", onPeerJoined);
   498→   498→      socket.off("call:accepted", onAccepted);
   499→   499→      socket.off("call:offer", onOffer);
   500→   500→      socket.off("call:answer", onAnswer);
   501→   501→      socket.off("call:ice", onIce);
   502→   502→      socket.off("call:peer-left", onPeerLeft);
   503→   503→      socket.off("call:ended", onEnded);
   504→   504→      socket.off("call:declined", onDeclined);
   505→   505→      socket.off("call:peer-state", onPeerState);
   506→   506→      socket.off("call:busy", onBusy);
   507→   507→    };
   508→   508→  }, [me?.id, createPeer, flushIce, endCallIfActive, setPeerState, toast, ringer]);
   509→   509→
   510→   510→  // Referencia de info de miembros accesible desde los handlers
   511→   511→  // (los handlers consultan el store directamente)
   512→   512→
   513→   513→  // Timbre mientras hay llamada entrante
   514→   514→  useEffect(() => {
   515→   515→    if (incomingCall) {
   516→   516→      ringer.start();
   517→   517→    } else {
   518→   518→      ringer.stop();
   519→   519→    }
   520→   520→  }, [incomingCall, ringer]);
   521→   521→
   522→   522→  const toggleMic = () => {
   523→   523→    const stream = localStreamRef.current;
   524→   524→    if (!stream) return;
   525→   525→    const next = !micOn;
   526→   526→    for (const t of stream.getAudioTracks()) t.enabled = next;
   527→   527→    setMicOn(next);
   528→   528→    const socket = getSocket();
   529→   529→    if (socket && call) {
   530→   530→      socket.emit("call:toggle", { conversationId: call.conversationId, audio: next });
